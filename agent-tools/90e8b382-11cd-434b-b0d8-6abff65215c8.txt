203 PMI_API_PUBLIC int PMI2_Job_Spawn(int count, const char *cmds[],

204 int argcs[], const char **argvs[],

205 const int maxprocs[],

206 const int info_keyval_sizes[],

207 const PMI2_keyval_t* info_keyval_vectors[],

208 int preput_keyval_size,

209 const PMI2_keyval_t preput_keyval_vector[],

210 char jobId[], int jobIdSize, int errors [])

211{

212 int pmi_errno= PMI2_SUCCESS;

213 int total_num_processes = 0;

214

215 /* Connect to the PM if we haven't already */

216 if (PMIi_InitIfSingleton() != 0)

217 return -1;

218

219 for (int i = 0; i < count; i++) {

220 total_num_processes += maxprocs[i];

221 }

222

223 struct PMIU_cmd pmicmd;

224 PMIU_msg_set_query_spawn(& pmicmd, USE_WIRE_VER, no_static,

225 count, cmds, maxprocs, argcs, argvs,

226 info_keyval_sizes, (const struct PMIU_token**) info_keyval_vectors,

227 preput_keyval_size, (const struct PMIU_token*) preput_keyval_vector);

228

229 pmi_errno= PMIU_cmd_get_response(PMI_fd, & pmicmd);

230 PMIU_ERR_POP(pmi_errno);

231

232 if (jobId && jobIdSize) {

233 const char * jid;

234 PMIU_CMD_GET_STRVAL(& pmicmd, "jobid", jid);

235 MPL_strncpy(jobId, jid, jobIdSize);

236 }

237

238 /* PMI2 does not return error codes, so we'll just pretend that means

239 * that it was going to send all `0's. */

240 for (int i = 0; i < total_num_processes; ++i) {

241 errors [i] = 0;

242 }

243

244 fn_exit:

245 return pmi_errno;

246 fn_fail:

247 goto fn_exit;

248

249}

443 static int parse_int_array(const char *str, int array[], int arraylen, int *outlen)

444{

445 int pmi_errno= PMI2_SUCCESS;

446

447 int rc;

448 int i = 0;

449 rc = sscanf(str, "%d", &array[i]);

450 PMIU_ERR_CHKANDJUMP1(rc != 1, pmi_errno, PMI2_ERR_OTHER,

451 "**intern %s", "unable to parse intarray");

452 ++i;

453 while ((str = strchr(str, ',')) && i < arraylen) {

454 ++str; /* skip over the ',' */

455 rc = sscanf(str, "%d", &array[i]);

456 PMIU_ERR_CHKANDJUMP1(rc != 1, pmi_errno, PMI2_ERR_OTHER,

457 "**intern %s", "unable to parse intarray");

458 ++i;

459 }

460

461 *outlen = i;

462

463 fn_exit:

464 return pmi_errno;

465 fn_fail:

466 goto fn_exit;

467}

650/*

651 * Singleton Init.

652 *

653 * MPI-2 allows processes to become MPI processes and then make MPI calls,

654 * such as MPI_Comm_spawn, that require a process manager (this is different

655 * than the much simpler case of allowing MPI programs to run with an

656 * MPI_COMM_WORLD of size 1 without an mpiexec or process manager).

657 *

658 * The process starts when either the client or the process manager contacts

659 * the other. If the client starts, it sends a singinit command and

660 * waits for the server to respond with its own singinit command.

661 * If the server start, it send a singinit command and waits for the

662 * client to respond with its own singinit command

663 *

664 * client sends singinit with these required values

665 * pmi_version= 

666 * pmi_subversion= 

667 *

668 * and these optional values

669 * stdio=[yes|no]

670 * authtype=[none|shared|]

671 * authstring= 

672 *

673 * server sends singinit with the same required and optional values as

674 * above.

675 *

676 * At this point, the protocol is now the same in both cases, and has the

677 * following components:

678 *

679 * server sends singinit_info with these required fields

680 * versionok=[yes|no]

681 * stdio=[yes|no]

682 * kvsname= 

683 *

684 * The client then issues the init command (see PMII_getmaxes)

685 *

686 * cmd=init pmi_version= pmi_subversion= 

687 *

688 * and expects to receive a

689 *

690 * cmd=response_to_init rc=0 pmi_version= pmi_subversion= 

691 *

692 * (This is the usual init sequence).

693 *

694 */
